---- Trigger: si al insertar no se manda "id" (NULL), se genera automáticamente con el formato RC-### usando el contador. Si sí se manda un id, se respeta tal cual (inserción manual) y el trigger no hace nada.--
CREATE TRIGGER trg_generar_id
AFTER INSERT ON Racks
FOR EACH ROW
WHEN NEW.id IS NULL
BEGIN
    UPDATE Racks
    SET id = 'RC-' || printf('%03d', (SELECT siguiente FROM contador WHERE id = 1))
    WHERE rowid = NEW.rowid;
 
    UPDATE contador SET siguiente = siguiente + 1 WHERE id = 1;
END;
